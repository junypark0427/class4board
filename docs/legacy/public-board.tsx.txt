import Link from 'next/link';
import { configured, database } from '@/lib/server';
import { categories, dateLabel, type Post } from '@/lib/shared';
export const dynamic = 'force-dynamic';
export default async function Board({ searchParams }: { searchParams: Promise<{ category?: string; page?: string }> }) {
  const params = await searchParams;
  const category = categories.find(c => c === params.category);
  const page = Math.min(10000, Math.max(1, Number.parseInt(params.page ?? '1') || 1));
  let posts: Post[] = []; let count = 0; let failed = false;
  const ready = configured();
  if (ready) {
    try {
      let query = database().from('posts').select('*', { count: 'exact' }).eq('status', 'approved').eq('visibility', 'public');
      if (category) query = query.eq('category', category);
      const result = await query.order('created_at', { ascending: false }).order('id').range((page - 1) * 12, page * 12 - 1);
      failed = Boolean(result.error); posts = result.data ?? []; count = result.count ?? 0;
    } catch { failed = true; }
  }
  const pageLink = (n: number) => `/?${new URLSearchParams({ ...(category ? { category } : {}), page: String(n) })}`;
  return <div className="container">
    <section className="intro"><div><p className="eyebrow">OUR CLASS, OUR VOICE</p><h1>사랑하는 4반<br/><span>익명 게시판</span><span className="heart" aria-hidden="true">♥</span></h1><p className="lead">말하기 어려웠던 생각도, 사소한 고마움도.<br/>이름 없이 편하게 남겨줘요.</p></div>
      <Link href="/write" className="button primary compose">＋ 마음 남기기</Link>
    </section>
    <div className="board-layout"><section className="board-main" aria-label="공개 게시글">
      <div className="section-heading"><h2>우리 반 이야기 <span className="count">{count}</span></h2><span className="subtle">최근 글부터</span></div>
      <nav className="filters" aria-label="게시글 분류"><Link className={!category ? 'chip selected' : 'chip'} href="/" aria-current={!category ? 'page' : undefined}>전체</Link>{categories.map(c => <Link href={`/?category=${encodeURIComponent(c)}`} key={c} className={category === c ? 'chip selected' : 'chip'} aria-current={category === c ? 'page' : undefined}>{c}</Link>)}</nav>
      {!ready ? <div className="empty"><span className="empty-mark" aria-hidden="true">…</span><h3>우리 반 게시판을 준비하고 있어요</h3><p>연결이 끝나면 여기에서 이야기를 나눌 수 있어요.<br/>지금은 글을 저장할 수 없어요.</p></div>
      : failed ? <div className="empty" role="alert"><h3>이야기를 불러오지 못했어요</h3><p>잠시 후 새로고침해주세요.</p><Link className="button secondary" href="/">다시 불러오기</Link></div>
      : posts.length === 0 ? <div className="empty"><span className="empty-mark" aria-hidden="true">＋</span><h3>{category ? '이 분류에는 아직 이야기가 없어요' : '첫 번째 이야기를 기다리고 있어요'}</h3><p>작은 생각 하나가 우리 반을 바꿀 수 있어요.<br/>공개 글은 관리자가 확인한 뒤 여기에 올라와요.</p><Link className="text-link" href="/write">첫 마음 남기기 ↗</Link></div>
      : <div className="post-list">{posts.map(post => <article className="post-card" key={post.id}><div className="post-meta"><span className="tag">{post.category}</span><time dateTime={post.created_at}>{dateLabel(post.created_at)}</time></div><h3><Link href={`/posts/${post.id}`}>{post.title}</Link></h3><p className="excerpt">{post.content}</p><div className="post-bottom"><span>익명의 4반 친구</span>{post.reply ? <span className="reply-indicator">답변이 도착했어요</span> : post.reply_requested ? <span>답변 기다리는 중</span> : null}</div></article>)}</div>}
      {ready && !failed && (page > 1 || count > page * 12) && <nav className="pagination" aria-label="페이지 이동">{page > 1 && <Link href={pageLink(page - 1)}>← 이전</Link>}<span>{page} 페이지</span>{count > page * 12 && <Link href={pageLink(page + 1)}>다음 →</Link>}</nav>}
    </section><aside className="sidebar"><div className="note"><p className="eyebrow">A LITTLE PROMISE</p><h2>편하게 말하고,<br/>다정하게 들어요.</h2><ol><li><strong>이름은 묻지 않아요</strong><span>학생 로그인 없이 마음을 전해요.</span></li><li><strong>공개 여부는 내가 정해요</strong><span>반 친구들에게, 또는 반장·부반장에게만.</span></li><li><strong>한 번 더 살펴봐요</strong><span>공개 글도 관리자의 승인 후 올라와요.</span></li></ol><Link className="text-link" href="/privacy">이용 안내 읽기 ↗</Link></div><p className="aside-caption">누군가를 향한 비난 대신,<br/>함께 바꾸고 싶은 점을 적어주세요.</p></aside></div>
  </div>;
}
